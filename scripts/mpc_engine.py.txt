# Masterpiece paperwork engine glue (runs inside Pyodide in the browser).
# Runs the unchanged company scripts as if from the command line and
# returns what they printed.
import sys, types, io, os, runpy, contextlib, json

# pdfminer imports `cryptography` (a compiled library) only to open password-
# protected PDFs. Company forms aren't encrypted, so a stand-in is enough.
def _stub(name):
    m = types.ModuleType(name)
    sys.modules[name] = m
    return m

class _NoCrypto:
    def __init__(self, *a, **k):
        raise RuntimeError("Password-protected PDFs aren't supported.")
    def __getattr__(self, k):
        raise RuntimeError("Password-protected PDFs aren't supported.")

for _n in ("cryptography", "cryptography.hazmat", "cryptography.hazmat.backends",
           "cryptography.hazmat.primitives", "cryptography.hazmat.primitives.ciphers"):
    _stub(_n)
sys.modules["cryptography.hazmat.backends"].default_backend = lambda: None
_c = sys.modules["cryptography.hazmat.primitives.ciphers"]
_c.Cipher = _NoCrypto
_c.algorithms = types.SimpleNamespace()
_c.modes = types.SimpleNamespace()

# reportlab imports Pillow at load time but only uses it for pictures. The
# scripts only draw text, so an empty stand-in is enough.
_pil = _stub("PIL"); _pil.__path__ = []; _pil.__version__ = "none"
_img = _stub("PIL.Image")
class _NoImage:
    def __init__(self, *a, **k):
        raise RuntimeError("Pictures aren't supported here.")
_img.Image = _NoImage
_img.open = _NoImage
_pil.Image = _img

SCRIPTS = "/mpc/scripts"

def run(script, argv):
    """Run SCRIPTS/script with argv; return JSON {code, stdout, stderr}."""
    path = os.path.join(SCRIPTS, script)
    out, err = io.StringIO(), io.StringIO()
    old_argv = sys.argv
    sys.argv = [path] + list(argv)
    code = 0
    try:
        with contextlib.redirect_stdout(out), contextlib.redirect_stderr(err):
            try:
                runpy.run_path(path, run_name="__main__")
            except SystemExit as e:
                code = e.code if isinstance(e.code, int) else (0 if e.code is None else 1)
            except Exception as e:
                import traceback
                traceback.print_exc()
                code = 1
    finally:
        sys.argv = old_argv
    if code == 0 and script == "build_contract.py":          # every sales contract gets the payment QR code
        try:
            for n in os.listdir(argv[-1]):
                if n.endswith("Sales Contract.pdf"):
                    add_pay_qr(os.path.join(argv[-1], n))
        except Exception as e:
            err.write("Payment QR code couldn't be added: %s\n" % e); code = 1
    return json.dumps({"code": code, "stdout": out.getvalue(), "stderr": err.getvalue()})

# Adds the Square, Venmo and Zelle payment QR codes to the bottom of a finished Sales Contract.
import io
PAY_URL = "https://checkout.square.site/merchant/ML9HVHFH1H8GA/checkout/G7QNAC6PN4X6D3JNPHWDZDF2"
VENMO_URL = "https://www.paypal.com/qrcodes/venmocs/ad306aa5-4ea1-4022-8dfd-c6f5a0bd7b6e?created=1718381611"
ZELLE_URL = "https://enroll.zellepay.com/qr-codes?data=eyJ0b2tlbiI6IjIwODg2NjYyMTEiLCJuYW1lIjoiVE9SSSBSVVVEIiwiYWN0aW9uIjoiUGF5bWVudCJ9"
CARLITO = "/usr/share/fonts/truetype/crosextra/Carlito-Regular.ttf"

# Chrome-style dinosaur drawn in the middle of the code (20 x 22 pixels).
DINO = ['...........########.', '..........##########', '..........##.#######', '..........##########', '..........##########', '..........##########', '..........#####.....', '..........########..', '#........#####......', '#......#######......', '##....##########....', '###..#########.#....', '##############......', '##############......', '.############.......', '..###########.......', '...#########........', '....#######.........', '.....###.##.........', '.....##...#.........', '.....#....#.........', '.....##...##........']

def add_pay_qr(path, have_square=False):
    from pypdf import PdfReader, PdfWriter
    from reportlab.pdfgen import canvas
    from reportlab.graphics.barcode.qr import QrCodeWidget
    from reportlab.graphics.shapes import Drawing
    from reportlab.graphics import renderPDF
    from reportlab.pdfbase import pdfmetrics
    from reportlab.pdfbase.ttfonts import TTFont
    w = PdfWriter(clone_from=path); pg = w.pages[0]
    W, H = float(pg.mediabox.width), float(pg.mediabox.height)
    size = 62.0; y = 20.0
    try: pdfmetrics.getFont("Carlito")
    except KeyError: pdfmetrics.registerFont(TTFont("Carlito", CARLITO))
    buf = io.BytesIO(); c = canvas.Canvas(buf, pagesize=(W, H))
    def code(url, cx, label, level, dino):
        q = QrCodeWidget(url, barLevel=level); q.barBorder = 0
        b = q.getBounds(); s = size / (b[2] - b[0])
        d = Drawing(size, size, transform=[s, 0, 0, s, 0, 0]); d.add(q)
        x = cx - size / 2
        renderPDF.draw(d, c, x, y)
        if dino:                                                  # Chrome-style dinosaur in the middle (20% of the width)
            px = size * 0.2 / 20; dw, dh = 20 * px, 22 * px; pad = px * 2
            dx, dy = x + (size - dw) / 2, y + (size - dh) / 2
            c.setFillColorRGB(1, 1, 1); c.rect(dx - pad, dy - pad, dw + 2 * pad, dh + 2 * pad, stroke=0, fill=1)
            c.setFillColorRGB(0, 0, 0)
            for r, row in enumerate(DINO):
                for k, ch in enumerate(row):
                    if ch == "#":
                        c.rect(dx + k * px, dy + (21 - r) * px, px * 1.02, px * 1.02, stroke=0, fill=1)
        c.setFont("Carlito", 9.5); c.setFillColorRGB(0.2, 0.2, 0.2)
        c.drawCentredString(cx, y - 12, label)
    # Square stays in the middle, exactly where it was first added; Venmo on the left, Zelle on the right.
    if not have_square:                                          # contracts from before Venmo/Zelle already have it
        code(PAY_URL, W / 2, "Scan to pay with Square", "H", True)
    code(VENMO_URL, W / 2 - 130, "Scan to pay with Venmo", "M", False)
    code(ZELLE_URL, W / 2 + 130, "Scan to pay with Zelle", "M", False)
    c.save()
    pg.merge_page(PdfReader(io.BytesIO(buf.getvalue())).pages[0])
    with open(path, "wb") as fh: w.write(fh)
