#!/usr/bin/env python3
"""Put the Square payment QR code on a sales contract that's already made
(for example one that was added to a job as a ready-made PDF).
Usage: add_pay_qr.py CONTRACT.pdf OUTPUT_DIR FILE_NAME"""
import sys, os, json, shutil
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from mpc_engine import add_pay_qr
src, out_dir, name = sys.argv[1], sys.argv[2], sys.argv[3]
os.makedirs(out_dir, exist_ok=True)
dest = os.path.join(out_dir, name)
shutil.copyfile(src, dest)
try:
    add_pay_qr(dest)
    print(json.dumps({"output": dest, "warnings": []}))
except Exception as e:
    print(json.dumps({"error": "The payment QR code couldn't be added: %s" % e})); sys.exit(1)
